# Mini_LM 依赖清单
#
# 安装（Windows 需要代理）：
#   set http_proxy=http://127.0.0.1:7897
#   set https_proxy=http://127.0.0.1:7897
#   pip install -r requirements.txt
#
# 环境：Python 3.13 + CUDA 12.6
# 实测可用组合（2026-09 验证）

# ---- 核心 ----
# torch 需要单独装 CUDA 版本，见 https://pytorch.org/get-started/locally/
# 本机用的是：pip install torch --index-url https://download.pytorch.org/whl/cu126
torch>=2.7.1
numpy>=2.0

# ---- 分词器 ----
# 字节级 BPE 训练。用 Rust 实现，速度约 51 万 tokens/s。
# 不需要 transformers —— 本项目自己实现模型，tokenizers 单独可用。
tokenizers>=0.20

# ---- 数据 ----
datasets>=3.0          # 下载 HF 数据集（备用通道）
huggingface_hub>=1.0   # HF 下载基础设施
pyarrow>=15.0          # 读写 parquet（魔塔和 HF 的语料多是 parquet）
modelscope>=1.40       # 魔塔社区 SDK（本项目主数据源）

# ---- 训练 ----
# 8-bit 优化器。**这是必需项不是优化项**：
# 普通 fp32 AdamW 的优化器状态要多占 1.9GB，会让总显存超过 6GB
# 触发 Windows 显存超售，训练慢 16 倍（实测 305 tok/s vs 4618 tok/s）
bitsandbytes>=0.45
safetensors>=0.4       # 权重的安全序列化格式（比 pickle 安全）
pyyaml>=6.0            # 读配置文件
tqdm>=4.66             # 进度条

# ---- 监控 ----
tensorboard>=2.15      # 训练曲线可视化

# ---- 可选 ----
# gradio>=5.0          # 网页版对话 demo（scripts/06_chat.py 不需要）
